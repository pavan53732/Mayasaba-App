def get_desktop_name(h):
    buf = ctypes.create_unicode_buffer(256)
    needed = ctypes.c_ulong()
    res = user32.GetUserObjectInformationW(h, 2, buf, 512, ctypes.byref(needed))
    return buf.value if res else None

def establish_foreground(hwnd):
    handle = int(hwnd)
    current_thread = kernel32.GetCurrentThreadId()
    foreground = user32.GetForegroundWindow()
    if foreground == handle:
        return True
    fg_thread = user32.GetWindowThreadProcessId(foreground, None) if foreground else 0
    app_thread = user32.GetWindowThreadProcessId(handle, None)
    
    attached_fg = False
    attached_app = False
    if fg_thread and fg_thread != current_thread:
        attached_fg = bool(user32.AttachThreadInput(current_thread, fg_thread, True))
    if app_thread and app_thread != current_thread and app_thread != fg_thread:
        attached_app = bool(user32.AttachThreadInput(current_thread, app_thread, True))
        
    user32.BringWindowToTop(handle)
    user32.SetForegroundWindow(handle)
    
    if attached_app:
        user32.AttachThreadInput(current_thread, app_thread, False)
    if attached_fg:
        user32.AttachThreadInput(current_thread, fg_thread, False)
        
    deadline = time.time() + 1.0
    while time.time() < deadline:
        if user32.GetForegroundWindow() == handle:
            return True
        time.sleep(0.05)
    return user32.GetForegroundWindow() == handle

def send_clean_drag(x1, y1, x2, y2, steps=60, delay=0.02):
    sw = user32.GetSystemMetrics(0) # SM_CXSCREEN
    sh = user32.GetSystemMetrics(1) # SM_CYSCREEN
    
    def norm_x(x): return int((x * 65535) / (sw - 1))
    def norm_y(y): return int((y * 65535) / (sh - 1))
    
    MOUSEEVENTF_MOVE = 0x0001
    MOUSEEVENTF_LEFTDOWN = 0x0002
    MOUSEEVENTF_LEFTUP = 0x0004
    MOUSEEVENTF_ABSOLUTE = 0x8000
    
    # 1. Position cursor over item
    user32.SetCursorPos(x1, y1)
    user32.mouse_event(MOUSEEVENTF_MOVE | MOUSEEVENTF_ABSOLUTE, norm_x(x1), norm_y(y1), 0, 0)
    # Wait longer than double-click interval (600ms) to ensure NO double-click occurs
    time.sleep(0.7)
    
    # 2. Press left button down to begin drag
    user32.mouse_event(MOUSEEVENTF_LEFTDOWN, 0, 0, 0, 0)
    time.sleep(0.15)
    
    # 3. Small initial motion to trigger drag detection
    user32.mouse_event(MOUSEEVENTF_MOVE, 8, 8, 0, 0)
    time.sleep(0.05)
    
    # 4. Interpolate across screen
    for i in range(1, steps + 1):
        cx = int(x1 + (x2 - x1) * (i / float(steps)))
        cy = int(y1 + (y2 - y1) * (i / float(steps)))
        user32.SetCursorPos(cx, cy)
        user32.mouse_event(MOUSEEVENTF_MOVE | MOUSEEVENTF_ABSOLUTE, norm_x(cx), norm_y(cy), 0, 0)
        time.sleep(delay)
        
    time.sleep(0.25)
    # 5. Release left button
    user32.mouse_event(MOUSEEVENTF_LEFTUP, 0, 0, 0, 0)
